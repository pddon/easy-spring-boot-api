```java
package com.pddon.framework.demo.easyapi.startup;

import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.context.annotation.ComponentScan;

import com.pddon.framework.easyapi.annotation.EnableEasyApi;

/**
 * 请求参数自动校验示例启动类。
 */
@SpringBootApplication
@ComponentScan(basePackages = "com.pddon.framework.demo.easyapi")
@EnableEasyApi
public class Startup {

    public static void main(String[] args) {
        SpringApplication.run(Startup.class, args);
    }
}
```

## 3 参数校验示例

> **实现说明（本次优化）**：框架的自动参数校验由`ValidateParamsInterceptor`完成，并仍然走原有拦截器链路。该拦截器通过`ObjectProvider<Validator>`延迟获取Spring容器中的`Validator`（通常是`LocalValidatorFactoryBean`），因此：
>
> * 自定义`ConstraintValidator`由Spring容器创建，可注入容器中的其他Bean；
> * 容器配置的`MessageInterpolator`、国际化`MessageSource`、分组校验行为保持不变；
> * 校验失败仍抛出`org.springframework.validation.BindException`，错误字段名、错误码与默认消息格式不变；
> * 脱离Spring容器手动`new ValidateParamsInterceptor()`时，回退到`Validation.buildDefaultValidatorFactory()`，行为与优化前一致；
> * 容器中存在多个`Validator` Bean时无法唯一确定，框架会回退到默认Validator并只告警一次；如依赖容器中的自定义`MessageInterpolator`，请保证容器中只有一个`Validator` Bean；
> * 本功能无需新增配置项；如需自定义校验器，按Spring Boot常规方式声明`ConstraintValidator` Bean即可。

### 3.1 校验系统参数

> easyapi内置了部分系统参数，也提供了定制和拓展系统参数的能力，使用方可以对这些系统参数做非空校验。

> 差异说明：原文档示例把系统参数校验、请求参数校验、响应参数校验分别放在`TestValidateSystemParamsController`、`TestValidateRequestParamsController`、`TestValidateResonseParamsController`三个类中，本模块的实际代码统一为`ValidateParamsController`（包`com.pddon.framework.demo.easyapi.validate`，路径前缀`/validate`）；`@RequiredParam`批量校验以及`@RequiredCountryCode`、`@RequiredCurrency`等注解在本模块未演示，系统参数仅演示`@RequiredLocale`。

* 接口源码`ValidateParamsController.java`

```java
package com.pddon.framework.demo.easyapi.validate;

import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

import com.pddon.framework.easyapi.annotation.RequiredLocale;
import com.pddon.framework.easyapi.context.RequestContext;

/**
 * 请求参数与响应参数自动校验示例。
 */
@RestController
@RequestMapping("/validate")
public class ValidateParamsController {

    /**
     * 框架会在方法调用前使用容器中的 Validator 校验该 DTO（默认分组）。
     * age 为奇数时会被自定义 EvenNumber 约束拦截。
     */
    @PostMapping("/checkBody")
    public ParamsCheckDto checkBody(@RequestBody ParamsCheckDto dto) {
        return dto;
    }

    /**
     * 使用 Spring 的 @Validated 指定 CreateGroup 分组进行校验。
     */
    @PostMapping("/create")
    public CreateDto create(@Validated(CreateGroup.class) @RequestBody CreateDto dto) {
        return dto;
    }

    /**
     * 使用 Spring 的 @Validated 指定 UpdateGroup 分组进行校验。
     */
    @PostMapping("/update")
    public UpdateDto update(@Validated(UpdateGroup.class) @RequestBody UpdateDto dto) {
        return dto;
    }

    /**
     * 缺少框架要求的系统参数 locale 时会被 ValidateSystemParamInterceptor 拦截。
     */
    @GetMapping("/requiredLocale")
    @RequiredLocale
    public String requiredLocale() {
        return RequestContext.getContext().getLocale();
    }

    /**
     * 正常返回，但响应参数不满足 @Max 约束，会被框架的响应校验拦截。
     */
    @GetMapping("/invalidResponse")
    public InvalidResponseDto invalidResponse() {
        return new InvalidResponseDto("张三", 200);
    }

    /**
     * 缺少必填的查询参数，由 Spring MVC 抛出参数缺失异常并交给框架统一处理。
     */
    @GetMapping("/missingQuery")
    public String missingQuery(@RequestParam String name) {
        return name;
    }
}
```

* 调用示例（默认端口8081，缺少`locale`系统参数时会被拦截）

```shell
curl "http://localhost:8081/validate/requiredLocale?appId=webApp&channelId=default&locale=zh_CN"
```

### 3.2 校验接口请求参数

> easyapi提供了自动校验接口请求参数的能力，包括域模型参数、基础参数等等。这将辅助接口使用者更清晰的知道错在了哪，规范了接口的输入，使得业务逻辑无需再手动校验参数的合法性。

* 域模型`ParamsCheckDto.java`（默认分组校验，`age`由自定义`@EvenNumber`约束校验）

```java
package com.pddon.framework.demo.easyapi.validate;

/**
 * 请求参数默认分组校验示例。
 */
public class ParamsCheckDto {

    private String name;

    @EvenNumber
    private Integer age;

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public Integer getAge() {
        return age;
    }

    public void setAge(Integer age) {
        this.age = age;
    }
}
```

* 自定义校验注解`EvenNumber.java`

```java
package com.pddon.framework.demo.easyapi.validate;

import java.lang.annotation.Documented;
import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;

import javax.validation.Constraint;
import javax.validation.Payload;

/**
 * 自定义校验注解示例：被标注的整数必须为偶数。
 */
@Documented
@Target({ElementType.FIELD, ElementType.PARAMETER})
@Retention(RetentionPolicy.RUNTIME)
@Constraint(validatedBy = EvenNumberValidator.class)
public @interface EvenNumber {

    String message() default "必须是偶数";

    Class<?>[] groups() default {};

    Class<? extends Payload>[] payload() default {};
}
```

* 自定义校验器`EvenNumberValidator.java`

```java
package com.pddon.framework.demo.easyapi.validate;

import javax.validation.ConstraintValidator;
import javax.validation.ConstraintValidatorContext;

public class EvenNumberValidator implements ConstraintValidator<EvenNumber, Integer> {

    @Override
    public boolean isValid(Integer value, ConstraintValidatorContext context) {
        return value == null || value % 2 == 0;
    }
}
```

* 分组校验DTO：`CreateDto.java`（`CreateGroup`分组）

```java
package com.pddon.framework.demo.easyapi.validate;

import javax.validation.constraints.NotBlank;

/**
 * 新增分组：name 必填。
 */
public class CreateDto {

    @NotBlank(groups = CreateGroup.class)
    private String name;

    @EvenNumber(groups = CreateGroup.class)
    private Integer age;

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public Integer getAge() {
        return age;
    }

    public void setAge(Integer age) {
        this.age = age;
    }
}
```

* 分组校验DTO：`UpdateDto.java`（`UpdateGroup`分组）

```java
package com.pddon.framework.demo.easyapi.validate;

import javax.validation.constraints.NotBlank;
import javax.validation.constraints.NotNull;

/**
 * 更新分组：id 与 name 必填。
 */
public class UpdateDto {

    @NotNull(groups = UpdateGroup.class)

    @NotBlank(groups = UpdateGroup.class)

    public Long getId() {

        this.id = id;
    }

    public String getName() {
        return name;
    }

    }
}
```

### 3.3 校验响应参数

> easyapi同样可以校验接口响应参数，用以规范接口开发者。

* 响应DTO源码`InvalidResponseDto.java`

```java
package com.pddon.framework.demo.easyapi.validate;

import javax.validation.constraints.Max;

/**
 * 用于演示响应参数校验失败的 DTO。
 */
public class InvalidResponseDto {

    private String name;

    @Max(value = 150, message = "年龄不能超过150")
    private Integer age;

    public InvalidResponseDto() {
    }

    public InvalidResponseDto(String name, Integer age) {
        this.name = name;
        this.age = age;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public Integer getAge() {
        return age;
    }

    public void setAge(Integer age) {
        this.age = age;
    }
}
```

> 差异说明：原文档示例的`TestValidateResonseParamsController`在本模块中对应`ValidateParamsController#invalidResponse`（`GET /validate/invalidResponse`），返回的`InvalidResponseDto.age`为200，超过`@Max(150)`，会被框架响应校验拦截；原示例中的`ParamsCheckDto`（account/email/likes）在本模块已不存在。

* 响应校验请求示例

```shell
curl "http://localhost:8081/validate/invalidResponse?appId=webApp&channelId=default"
```
